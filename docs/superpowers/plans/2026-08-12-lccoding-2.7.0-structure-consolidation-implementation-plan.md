### Exact hash refresh

```powershell
@'
from pathlib import Path
import hashlib, json
root = Path.cwd()
files = sorted(
    path for path in root.rglob("*")
    if path.is_file()
    and ".git" not in path.relative_to(root).parts
    and ".codex" not in path.relative_to(root).parts
    and "__pycache__" not in path.relative_to(root).parts
    and path.suffix != ".pyc"
    and path.name != "FILE_HASHES.json"
)
manifest = {
    path.relative_to(root).as_posix(): hashlib.sha256(path.read_bytes()).hexdigest()
    for path in files
}
with (root / "FILE_HASHES.json").open("w", encoding="utf-8", newline="\n") as stream:
    json.dump(manifest, stream, ensure_ascii=False, indent=2)
    stream.write("\n")
'@ | python -
```

### Safe external BI runner

Use this before every Node, Rust, Playwright, or packaging verification:

```powershell
$repo = [IO.Path]::GetFullPath((Resolve-Path .).Path)
$tmpRoot = [IO.Path]::GetFullPath('D:\LCcoding\.codex\.tmp')
$fixedRunner = [IO.Path]::GetFullPath('D:\LCcoding\.codex\.tmp\lccoding-270-runner')
$fixedCargo = [IO.Path]::GetFullPath('D:\LCcoding\.codex\.tmp\lccoding-270-cargo-target')
$runner = [IO.Path]::GetFullPath($fixedRunner)
$cargoTarget = [IO.Path]::GetFullPath($fixedCargo)
foreach($pair in @(@($runner,$fixedRunner),@($cargoTarget,$fixedCargo))){
  if($pair[0] -cne $pair[1]){ throw 'temporary target is not the fixed allowed path' }
  $relative = [IO.Path]::GetRelativePath($tmpRoot,$pair[0])
  if([IO.Path]::IsPathRooted($relative) -or $relative -eq '..' -or $relative.StartsWith("..$([IO.Path]::DirectorySeparatorChar)")){
    throw 'temporary target escaped D:\LCcoding\.codex\.tmp'
  }
}
foreach($target in @($runner,$cargoTarget)){
  if(Test-Path -LiteralPath $target){ Remove-Item -LiteralPath $target -Recurse -Force }
}
New-Item -ItemType Directory -Path $runner | Out-Null
$inputs = @(git ls-files --cached --others --exclude-standard)
$forbiddenInput = '(^|/)(node_modules|dist|target|gen|test-results|html-report|coverage|screenshots?|playwright-report)(/|$)'
foreach($relative in $inputs){
  $normalized = $relative.Replace('\','/')
  if($normalized -match $forbiddenInput){ throw "generated input forbidden: $normalized" }
  $source = Join-Path $repo $relative
  if(Test-Path -LiteralPath $source -PathType Leaf){
    $destination = Join-Path $runner $relative
    New-Item -ItemType Directory -Path (Split-Path $destination) -Force | Out-Null
    Copy-Item -LiteralPath $source -Destination $destination
  }
}
$runnerBi = Join-Path $runner 'lc-coding/bi'
$runnerTauri = Join-Path $runnerBi 'src-tauri'
Push-Location $runnerBi
npm ci --ignore-scripts
$env:LCCODING_BI_DIST = (Join-Path $runner 'frontend-dist').Replace('\','/')
npm run build
Pop-Location
$relativeDist = [IO.Path]::GetRelativePath($runnerTauri,$env:LCCODING_BI_DIST).Replace('\','/')
$env:TAURI_CONFIG = @{build=@{frontendDist=$relativeDist}} | ConvertTo-Json -Compress
$env:CARGO_TARGET_DIR = $cargoTarget
$env:BI_OWNER_REVIEW_DIR = Join-Path $runner 'visual-review'
```
